set -euo pipefail

app="inhouse-bot"
out="/root/.pm2/logs/$app-out.log"
err="/root/.pm2/logs/$app-error.log"

echo "=== Checking before making changes ==="
date -Is

app_pid=$(pm2 pid "$app")
[[ "$app_pid" =~ ^[0-9]+$ && "$app_pid" != 0 ]] || {
  echo "STOP: community app is not running as expected."
  exit 1
}

# Verify the parser JAR exists where this app expects it.
test -r "/proc/$app_pid/cwd/odota-parser/target/stats-0.1.0.jar" || {
  echo "STOP: parser JAR not found in the app's working directory."
  exit 1
}

pid=$(fuser -n tcp 5600 2>/dev/null | xargs || true)
[[ "$pid" =~ ^[0-9]+$ ]] || {
  echo "STOP: expected exactly one parser process on port 5600."
  exit 1
}

[[ "$(ps -p "$pid" -o comm= | xargs)" == "java" &&
   "$(ps -p "$pid" -o ppid= | xargs)" == "1" ]] || {
  echo "STOP: listener is no longer an orphaned Java process."
  exit 1
}

tr '\0' '\n' < "/proc/$pid/cmdline" |
  grep -Eq '(^|/)stats-0\.1\.0\.jar$' || {
    echo "STOP: listener does not match the expected parser JAR."
    exit 1
  }

ps -p "$pid" -o pid,ppid,lstart,comm
out_start=$(( $(wc -l < "$out") + 1 ))
err_start=$(( $(wc -l < "$err") + 1 ))

# Attempt to restore the community app if a later step fails.
resume=1
